# products schema

- name: string, required, min-len 2
- desc: string, max-len 500
- price: number, min 0
- taxRate: number, min 0
- deliveryCharges: number, min 0
- rating: number min 0, max 5
- images: [string] min 1 entry
- category: enum, required
- gender: enum, required
- stock: number
- sizes: [string]
- colors: [string]
